PENUTUP DI BAGIAN BAWAH index.html (setelah semua kode game, sebelum tag terakhir) ========================================================= --> penutup ============================================================ */ (function() { 'use strict'; /* ----------------------------------------------------------- 1. FIX CANVAS SIZE — otomatis fit area yang tersedia ----------------------------------------------------------- */ function resizeGameCanvas() { const container = document.getElementById('game-container'); const canvas = document.getElementById('game-canvas'); if (!container || !canvas) return; const containerW = container.clientWidth; const containerH = container.clientHeight; if (containerW <= 0 || containerH <= 0) return; // Hitung skala terbesar yang masih fit (aspect ratio dipertahankan) // Canvas internal game biasanya 400x300 atau sesuai CELL_SIZE * GRID // Ambil ukuran internal dari canvas attribute const internalW = canvas.width || 400; const internalH = canvas.height || 300; const scaleX = containerW / internalW; const scaleY = containerH / internalH; const scale = Math.min(scaleX, scaleY, 1); // max 1x supaya tidak blur // Jika canvas dirender dengan width/height attribute, // set style size saja (tidak ubah internal resolution) canvas.style.width = Math.floor(internalW * scale) + 'px'; canvas.style.height = Math.floor(internalH * scale) + 'px'; } /* ----------------------------------------------------------- 2. FIX PORTRAIT OVERLAY — izinkan bermain di portrait Hapus/comment fungsi ini jika mau tetap landscape only ----------------------------------------------------------- */ function allowPortraitPlay() { const overlay = document.getElementById('portrait-overlay'); if (!overlay) return; // Sembunyikan overlay portrait agar bisa main di semua orientasi // Comment 2 baris di bawah jika mau tetap paksa landscape overlay.style.display = 'none'; overlay.classList.remove('visible'); } /* ----------------------------------------------------------- 3. FIX SCREEN-BOX MAX HEIGHT — pakai dvh jika tersedia ----------------------------------------------------------- */ function fixScreenBoxHeight() { const boxes = document.querySelectorAll('.screen-box'); const vh = window.innerHeight; boxes.forEach(box => { box.style.maxHeight = (vh - 24) + 'px'; }); // Fix menu juga const menu = document.getElementById('screen-menu'); if (menu) { menu.style.maxHeight = vh + 'px'; } } /* ----------------------------------------------------------- 4. FIX DPAD POSITION — pastikan tidak keluar layar ----------------------------------------------------------- */ function fixDpad() { const dpad = document.getElementById('dpad'); if (!dpad) return; const W = window.innerWidth; const H = window.innerHeight; const dpadW = dpad.offsetWidth || 130; const dpadH = dpad.offsetHeight || 130; // Pastikan dpad tidak keluar layar kanan/bawah const maxLeft = W - dpadW - 8; const maxBottom = H - dpadH - 8; const currentLeft = parseInt(dpad.style.left || '10'); const currentBottom = parseInt(dpad.style.bottom || '14'); if (currentLeft > maxLeft) { dpad.style.left = Math.max(8, maxLeft) + 'px'; } if (currentBottom > maxBottom) { dpad.style.bottom = Math.max(8, maxBottom) + 'px'; } } /* ----------------------------------------------------------- 5. FIX MENU LOGO — pastikan tidak overflow secara vertikal ----------------------------------------------------------- */ function fitMenuToScreen() { const screenMenu = document.getElementById('screen-menu'); if (!screenMenu || !screenMenu.classList.contains('active')) return; const H = window.innerHeight; // Jika semua konten lebih tinggi dari layar, enable scroll screenMenu.style.overflowY = 'auto'; screenMenu.style.justifyContent = H < 500 ? 'flex-start' : ''; } /* ----------------------------------------------------------- 6. GABUNGKAN SEMUA — jalankan saat resize & orientasi berubah ----------------------------------------------------------- */ function applyAllFixes() { allowPortraitPlay(); fixScreenBoxHeight(); fixDpad(); fitMenuToScreen(); resizeGameCanvas(); } // Jalankan setelah DOM siap if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', applyAllFixes); } else { applyAllFixes(); } // Jalankan saat ukuran layar berubah (orientasi, keyboard virtual, dll) window.addEventListener('resize', applyAllFixes); window.addEventListener('orientationchange', function() { // Tunggu sebentar agar browser selesai update dimensi setTimeout(applyAllFixes, 300); }); // Jalankan juga saat layar aktif (pindah screen) // Intercept showScreen jika ada fungsinya const _origShowScreen = window.showScreen; if (typeof _origShowScreen === 'function') { window.showScreen = function(id) { _origShowScreen(id); setTimeout(applyAllFixes, 50); }; } else { // Fallback: observe class changes pada .screen elements const screens = document.querySelectorAll('.screen'); const observer = new MutationObserver(function(mutations) { mutations.forEach(function(m) { if (m.attributeName === 'class') { setTimeout(applyAllFixes, 50); } }); }); screens.forEach(function(s) { observer.observe(s, { attributes: true }); }); } console.log('[ResponsiveFix] Applied successfully'); })();
